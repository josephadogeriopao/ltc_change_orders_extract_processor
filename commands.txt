# --- ONE-TIME SETUP ---
# Create the environment
python -m venv .venv

# Activate (Windows)
.venv\Scripts\activate

# Activate (Mac/Linux)
source .venv/bin/activate

# --- INSTALLING PACKAGES ---
# OR Install everything from a shared list
pip install -r requirements.txt

# --- SAVING YOUR PACKAGES TO FILE ---
# Update whenever you install something new
pip freeze > requirements.txt

# --- PACKAGING AND DISTRIBUTING APP ---
python -m PyInstaller --noconsole --onefile --name "change_orders_generator" --icon "assets/images/app_icon.ico" --add-data "assets;assets" --paths "src" --collect-all customtkinter src/app.py

# -- THE SYSTEM CLEAN CACHED ASSETS LIKE REPLACED IMAGES ETC. --
# Delete Your Local PyInstaller Cache
# To make sure it isn't reuse-loading an old compiled .ico format internally, append the --clean flag to your command:
python -m PyInstaller --clean --noconsole --onefile --name


# --- RUNNING TEST SUITES ---
python -m unittest test_labels_engine.py
